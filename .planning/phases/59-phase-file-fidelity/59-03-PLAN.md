# Plan 59-03 — FIDEL-04: Validator Verdict Flooring for manifest_violation

## Objective

Extend `manifestCheck` (and its CLI wrapper `_runManifestCheckCli`) so that a
`manifest_violation` action floors the exit code at `2` (the `gaps_found`
convention already used throughout the codebase) rather than `1`. A verdict
that exits `2` cannot be treated as `passed` by the workflow without an
explicit atomization-reconciliation step. The `atomization_reconciliation`
flag must appear in the violation report JSON before the workflow can consider
the task complete.

## Verification Criteria

1. `node get-shit-done/bin/gsd-tools.cjs manifest-check ...` exits `2` when
   `manifest_violation` is detected (currently exits `1`).
2. The violation report JSON written to disk includes an
   `atomization_reconciliation_required: true` field.
3. `node --test tests/59-03-fidel04-verdict-floor.test.cjs` exits 0.

## must_haves

- The change is to `_runManifestCheckCli` only — `manifestCheck()` itself
  returns a result object; the exit code is a CLI concern. Specifically,
  L1098: `if (result.action === 'halt' || result.action === 'halt_orchestrator_owned') { return 1; }`
  must become `return 2` for `halt` (the manifest_violation path). Keep
  `halt_orchestrator_owned` at exit 1 (that is a different violation class
  with its own escalation path).
- The violation report JSON (written to disk in `manifestCheck`) must include
  `atomization_reconciliation_required: true` when `action === 'halt'` so the
  operator workflow can verify the reconciliation field is present before
  allowing the task to proceed.
- `action === 'warn'` continues to exit 0 (not a hard stop).
- `action === 'pass'` continues to exit 0.
- `action === 'halt_orchestrator_owned'` continues to exit 1 (not changed —
  orchestrator-owned violations have a different escalation path).
- Backward-compat: all callers that check `result.action === 'halt'` and
  handle non-zero exit codes are unaffected — they already treat non-zero as
  a stop signal. The exit code change from 1→2 for `halt` is the only
  behavioral delta.
- This plan MUST land after 59-01 (FIDEL-01 extends `manifestCheck`'s glob
  rejection) since 59-01 modifies the same function. Depends-on: 59-01.

    'use strict';
    const { test } = require('node:test');
    const assert = require('node:assert/strict');
    const path = require('path');
    const fs = require('fs');
    const os = require('os');
    const { spawnSync, execFileSync } = require('child_process');

    const REPO_ROOT = path.resolve(__dirname, '..');
    const TOOLS = path.join(REPO_ROOT, 'get-shit-done', 'bin', 'gsd-tools.cjs');

    // Helper: run manifest-check CLI and return the result
    function runManifestCheck({ phase, wave, taskId, filesExpectedYaml, before, after }) {
      const tmp = path.join(os.tmpdir(), `fe-${Date.now()}.yaml`);
      fs.writeFileSync(tmp, filesExpectedYaml, 'utf-8');
      try {
        const r = spawnSync('node', [
          TOOLS, 'manifest-check',
          '--phase', phase,
          '--wave', wave,
          '--task-id', taskId,
          '--files-expected', tmp,
          '--before', before,
          '--after', after,
        ], {
          cwd: REPO_ROOT,
          encoding: 'utf-8',
          timeout: 15000,
          env: { ...process.env, GSD_AMAUTA_NO_AUTO_START: '1' },
        });
        return r;
      } finally {
        try { fs.unlinkSync(tmp); } catch {}
      }
    }

    test('FIDEL-04: manifest-check exits 2 on halt (manifest_violation)', async (t) => {
      // We need a before/after SHA pair where actual diff differs from expected.
      // Use the same SHA for both (empty diff) but declare a file to create
      // that does not exist — this produces a missing_creates violation → halt action.

      // Get current HEAD SHA
      const headSha = execFileSync('git', ['rev-parse', 'HEAD'], { cwd: REPO_ROOT, encoding: 'utf-8' }).trim();

      // Declare a create that did not happen
      const filesExpectedYaml = `modify: []\ncreate:\n  - nonexistent-file-that-will-never-exist.txt\ndelete: []`;

      const r = runManifestCheck({
        phase: '59',
        wave: '1',
        taskId: '59-test-halt',
        filesExpectedYaml,
        before: headSha,
        after: headSha,
      });

      // When files_expected.create lists a file that is not in the actual diff,
      // that is a missing_creates violation → action = 'halt' → exit 2 (FIDEL-04)
      assert.equal(r.status, 2, `Expected exit 2 on manifest_violation, got ${r.status}. stderr: ${r.stderr}`);

      // _runManifestCheckCli writes {ok, reportPath, action} to stdout — the FULL
      // violation report (with atomization_reconciliation_required) is the file
      // at reportPath, NOT stdout. Read the on-disk report to assert the field.
      let cliOut;
      try { cliOut = JSON.parse(r.stdout); } catch { cliOut = null; }
      assert.ok(cliOut, 'CLI should output {ok, reportPath, action} JSON');
      assert.equal(cliOut.action, 'halt', 'action should be halt');
      assert.ok(cliOut.reportPath && fs.existsSync(cliOut.reportPath), `report file should exist at ${cliOut && cliOut.reportPath}`);
      const report = JSON.parse(fs.readFileSync(cliOut.reportPath, 'utf-8'));
      assert.equal(report.atomization_reconciliation_required, true, 'on-disk report must carry atomization_reconciliation_required:true on halt');
    });

    test('FIDEL-04: manifest-check exits 0 on pass (no violations)', async (t) => {
      const headSha = execFileSync('git', ['rev-parse', 'HEAD'], { cwd: REPO_ROOT, encoding: 'utf-8' }).trim();
      // No expected creates/modifies/deletes + same SHA = no diff = pass
      const filesExpectedYaml = `modify: []\ncreate: []\ndelete: []`;

      const r = runManifestCheck({
        phase: '59',
        wave: '1',
        taskId: '59-test-pass',
        filesExpectedYaml,
        before: headSha,
        after: headSha,
      });

      assert.equal(r.status, 0, `Expected exit 0 on pass, got ${r.status}`);
      let report;
      try { report = JSON.parse(r.stdout); } catch { report = null; }
      assert.ok(report, 'Should output JSON');
      assert.equal(report.action, 'pass');
    });

    test('FIDEL-04: halt report contains atomization_reconciliation_required: true', async (t) => {
      const headSha = execFileSync('git', ['rev-parse', 'HEAD'], { cwd: REPO_ROOT, encoding: 'utf-8' }).trim();
      const filesExpectedYaml = `modify: []\ncreate:\n  - will-not-exist-${Date.now()}.txt\ndelete: []`;

      const r = runManifestCheck({
        phase: '59',
        wave: '1',
        taskId: '59-test-arec',
        filesExpectedYaml,
        before: headSha,
        after: headSha,
      });

      let cliOut;
      try { cliOut = JSON.parse(r.stdout); } catch { cliOut = null; }
      assert.ok(cliOut && cliOut.reportPath && fs.existsSync(cliOut.reportPath), 'CLI stdout must carry reportPath to an existing on-disk report');
      const report = JSON.parse(fs.readFileSync(cliOut.reportPath, 'utf-8'));
      assert.strictEqual(report.atomization_reconciliation_required, true);
    });
